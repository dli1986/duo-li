class Head(nn.Module):
    """ one head of self-attention """

    def __init__(self, head_size):
        super().__init__()
        self.key   = nn.Linear(n_embd, head_size, bias=False)
        self.query = nn.Linear(n_embd, head_size, bias=False)
        self.value = nn.Linear(n_embd, head_size, bias=False)
        self.register_buffer('tril', torch.tril(torch.ones(block_size, block_size)))
        self.dropout = nn.Dropout(0.1)

    def forward(self, x):
        B,T,C = x.shape
        k = self.key(x)   # (B,T,head_size)          -- ①
        q = self.query(x) # (B,T,head_size)          -- ②
        v = self.value(x) # (B,T,head_size)          -- ③  (computed here, used in ⑦)
        wei = q @ k.transpose(-2, -1) * head_size**-0.5  # (B,T,T) -- ④ independence BREAKS
        wei = wei.masked_fill(self.tril[:T, :T] == 0, float('-inf'))  # -- ⑤
        wei = F.softmax(wei, dim=-1)  # -- ⑥
        wei = self.dropout(wei)
        out = wei @ v  # (B,T,head_size)  -- ⑦ content actually mixes
        return out
PyTorch internals: nn.Linear(n_embd, head_size, bias=False) stores a weight of shape (head_size, n_embd) (PyTorch stores Linear weights transposed relative to the math) — calling it computes x @ weight.T internally. register_buffer('tril', ...) is how you attach a tensor to the module that moves with .to(device) and saves with state_dict(), but is excluded from .parameters() — the optimizer never touches it, exactly right for a fixed causal mask.

# single head self attention
class Head(nn.Module):
    """ one head of self-attention """

    def __init__(self, head_size):
        super().__init__()
        # 传承前面的input的维度，B,T,C(vocab_size-->n_embd)，将n_embd维的输入映射到head_size维的空间中，得到query、key、value向量
        # 收敛的时候，head_size维的query(16)、key(16)、value(16)向量会学会捕捉输入数据中的某些特定模式或关系，这些模式或关系对于模型的预测任务是有用的
        # 结束后收敛回n_embd维的空间中,基于head_size维的query、key、value向量进行计算，得到一个新的n_embd维的表示，这个表示包含了输入数据中捕捉到的特定模式或关系的信息
        # 最后在最后一层的lm_head中将n_embd维的表示映射回vocab_size维的空间中，得到每个位置上每个字符的预测分数
        # 这里是真正的需要更新的参数，W_key,W_query,W_value是在nn.Linear中定义并被记录为参数的
        
        # self.key/query/value 是可学习的线性投影层。
        # PyTorch 中每个 Linear 的 weight shape 是 (head_size, n_embd)，
        # 但数学上等价于把输入 x 右乘一个 (n_embd, head_size) 的矩阵。

        self.key = nn.Linear(n_embd, head_size, bias=False)
        self.query = nn.Linear(n_embd, head_size, bias=False)
        self.value = nn.Linear(n_embd, head_size, bias=False)
        # 基于block size和tril矩阵的mask，确保每个位置只能关注到它自己和它之前的位置，防止模型在训练过程中看到未来的信息
        
        # tril 是固定的 causal mask，不是可学习参数。
        # register_buffer 让它跟随模型保存和移动设备，但不会被 optimizer 更新。

        self.register_buffer('tril', torch.tril(torch.ones(block_size, block_size)))
        self.dropout = nn.Dropout(0.1)
    
    # 前向传播函数，输入x的形状是(B,T,C)，输出是(B,T,head_size), 这里的C是n_embd
    def forward(self, x):
        B,T,C = x.shape
        # 通过线性变换得到key、query、value向量
        # 这里有一个核心注意点是，在前向传播的时候这里的k,q,wei都是中间值，并不是参数
        # 它们是动态计算得到的，基于输入数据x通过线性变换得到的，这些中间值会随着输入数据的不同而变化，并且在训练过程中会不断更新，以学习到更好的模式或关系
        # 真正的参数是self.key、self.query、self.value这三个线性层的权重，这些权重会在训练过程中通过反向传播进行更新，以学习到更好的模式或关系
        # 它们的形状是(head_size, n_embd)（PyTorch Linear 的 weight 约定），是固定的，而k,q,wei的形状是(B,T,head_size)和(B,T,T)，是动态的，随着输入数据的不同而变化

        # q/k/v 是由输入 x 和可学习参数动态计算出的中间激活，不是模型参数。
        k = self.key(x)   # (B,T,head_size)
        q = self.query(x) # (B,T,head_size)
        # 计算注意力分数
        # 这里的tril矩阵应用还是和平均概率分布的mask一样，确保每个位置只能关注到它自己和它之前的位置，防止模型在训练过程中看到未来的信息
        # 不同的是通过引入的k和q让模型在不看未来的前提下，不再是平均地关注之前的所有位置，而是根据输入数据中的模式或关系，学习到在不同位置上应该关注哪些之前的位置，这些位置对于模型的预测任务是有用的

        
        # attention logits: 每个 token 的 query 与所有 token 的 key 做相似度计算。
        # 缩放因子用 1/sqrt(head_size)（不是 1/sqrt(n_embd)）。

        # 真正排除未来 token 的不是 transpose，而是后面的 causal mask
        
        # q @ k.transpose(-2, -1)

        # (B, T, head_size) @ (B, head_size, T)
        # = (B, T, T)

        wei = q @ k.transpose(-2, -1) * head_size**-0.5 # (B,T,head_size) @ (B,head_size,T) --> (B,T,T)
        # 这个tril矩阵的形状抛开batch size的话，是一个T*T的矩阵（T 是本次输入的实际序列长度，不是 block_size 本身——
        # block_size 只是 register_buffer 时开的最大容量上限，这里用 self.tril[:T, :T] 按需切片）

        # causal mask: 禁止当前位置关注未来 token（只挡住 t'>t，t'<=t 含自身都保留）
        # 注意：这里是"多位置并行"场景（一次前向同时算 T 个位置），才需要这行显式 mask；
        # 如果是 KV Cache 的增量解码（每步只有 1 个新 query 对历史 cache），cache 里天生没有未来，不需要这行
        wei = wei.masked_fill(self.tril[:T, :T] == 0, float('-inf')) # (B,T,T) -> (B,T,T)

        
        # softmax 后得到动态 attention distribution。
        # 这里的 wei 可称为 attention weights，但它不是 trainable parameters。

        wei = F.softmax(wei, dim=-1) # (B,T,T)
        wei = self.dropout(wei)
        v = self.value(x) # (B,T,head_size)
        # B,T,T @ B,T,head_size --> B,T,head_size

        # 用 attention distribution 对 value 做加权聚合。
        out = wei @ v 
        return out # (B,T,T) @ (B,T,head_size) --> (B,T,head_size)

note 0487e841

Attention 内部：从"独立"到"关联"的精确临界点

这是理解 attention 最容易含糊带过的地方——"跟其他 token 有关系"这句话太笼统， 下面按 Head.forward 里的每一行代码标出独立性在哪一行被打破、又在哪一行被约束。

def forward(self, x):              # x: (B, T, C)，此时每个 (b,t) 位置的向量互不相干（独立）
    B, T, C = x.shape

    k = self.key(x)                 # 第①行 —— 仍然独立
    q = self.query(x)               # 第②行 —— 仍然独立
    v = self.value(x)               # 第③行 —— 仍然独立

    wei = q @ k.transpose(-2, -1) * C**-0.5   # 第④行 —— 独立被打破的临界点！
    wei = wei.masked_fill(self.tril[:T, :T] == 0, float('-inf'))  # 第⑤行 —— 强制约束关联范围
    wei = F.softmax(wei, dim=-1)    # 第⑥行 —— 归一化，此刻只是"关系强度"，还没混合内容

    out = wei @ v                   # 第⑦行 —— 真正发生"内容混合"的地方
    return out

逐行拆解独立性状态

① ② ③ k/q/v = self.xxx(x)——仍然独立

nn.Linear(n_embd, head_size) 是一个**逐位置（pointwise）**操作：对 (B,T,C) 的每一个 (b,t) 向量， 独立地做同一个矩阵乘法 x[b,t,:] @ W，t 和 t' 之间没有任何计算交叉。 B 和 T 在这一步都还是并行独立的——跟 embedding 查表那一步的独立性质完全一样。

④ wei = q @ k.transpose(-2,-1)——独立被打破的确切时刻

这是全部计算里唯一让 T 个位置开始互相"看见"对方的一行。 q 的形状是 (B,T,head_size)，k.transpose 后是 (B,head_size,T)， 矩阵乘法迫使 q 的每一行（每个位置的 query）去跟 k 的每一列（每个位置的 key）做点积， 结果 wei[b,t,t'] 表示"位置 t 与位置 t' 的相似度"——从这一行开始，位置 t 的计算结果依赖于其他所有位置 t'。

（B 维度自始至终没有参与这次矩阵乘法的交叉，torch.bmm 风格的批量矩阵乘法只在每个 batch 内部做，batch 之间永远独立。）

⑤ masked_fill(tril==0, -inf)——约束关联范围，不是恢复独立

第④行让位置 t 能看到所有位置（包括未来），这一行强制把 t' > t 的关联度设为 -inf。 这不是"重新变回独立"，而是限定了"能关联到谁"的范围：t 只能跟 t' <= t 产生关联， t' > t 的关联强度归零（经过 softmax 后变成概率 0）。

⑥ softmax(wei, dim=-1)——只是归一化"关系强度"，内容还没被混合

这一步把 wei[b,t,:] 变成一行合法的概率分布（加起来等于 1）， 但此时 wei 里存的仍然只是"位置 t 应该多大程度参考位置 t'"这个权重， value 的实际内容（v）还没有被加权求和，混合尚未发生。

⑦ out = wei @ v——内容真正发生混合的地方

wei (B,T,T) 与 v (B,T,head_size) 相乘，out[b,t,:] = 对所有允许的 t'，按 wei[b,t,t'] 加权求和 v[b,t',:]。 到这一步，位置 t 的输出向量里才真正融入了其他位置（t' <= t）的内容信息，不再是它自己独立的向量。

一句话总结独立→关联的转变

x, k, q, v          : B 独立, T 内部各位置也独立（逐位置线性变换）
        ↓ q @ k.T   :  ← 这一步引入 T 内部位置间的关联（B 依然独立）
wei (masked+softmax) : T 内部关联被限定在"只能看过去"，且已归一化为权重
        ↓ wei @ v   :  ← 这一步才把关联转化为真正的内容混合
out                 : 每个位置的向量已经不再独立，包含了它能看到的历史信息

B 这个维度在整个 attention 计算中永远保持独立——不管哪一行代码，batch 之间都不会发生任何交叉计算， 这也是为什么"batch 之间独立"是一条从 embedding 到最终输出都成立的不变量。

note 44e0837e

精确定位：两次矩阵乘法各自消掉了哪个维度

一个常见的表述偏差是笼统地说"C 在乘法里被消除了"，需要把 C 和 head_size 分开，精确到位。

1. q、k 天然携带内容+位置信息：q = x @ W_q、k = x @ W_k，两者都是从同一个 x（tok_emb + pos_emb）分别投影出来的， 所以每个位置 t 的 q[t]、k[t] 里都编码了"这个位置的内容是什么 + 排第几"。

2. 精确的维度消除顺序：

第一次乘法：x @ W        (B,T,C) @ (C,head_size) → (B,T,head_size)
            ↑ 消除的是 C，结果压缩成 head_size 维

此时 q、k 已经不是 C 维了，而是 head_size 维——
"内容+位置"信息已经被压缩进了这个更小的 head_size 维向量里

第二次乘法：q @ k.T      (B,T,head_size) @ (B,head_size,T) → (B,T,T)
            ↑ 消除的是 head_size，不是 C（C 在上一步就已经不存在了）

更准确的说法是：C 在第一次乘法里就已经被"用掉"压缩成了 head_size；第二次乘法消除的是这个 head_size， 产出的 wei[b,t,t'] 是一个标量（相似度分数），不再携带任何维度的语义细节—— "这里 C 不重要"这个直觉是对的，只是发生的时间点在更早的那一步。

3. 最后一步 out = wei @ v：wei 只是"关注权重"（标量集合），真正的内容（v）在这一步才被加权带入， 位置 t 的输出向量从此不再独立，融合了它能看到的历史位置的信息。

一句话总结：两次矩阵乘法各自消掉一个维度（C→head_size→标量），信息逐步从"每个位置自己的表示"压缩成"位置间的关系强度"，最后再用这个关系强度去加权聚合 v，完成真正的内容混合。

note 5de99ef7

补充理解：C 不是"不重要"，而是"被消耗掉产出了关系分数"——v 才是语义的真正载体

前面说"C 在第二次乘法里不重要"容易被误读成"C 没有意义"，需要纠正为更精确的版本：

C 从一开始就是有意义的：token_embedding_table + position_embedding_table 两次查表， 共同赋予了任意 token ID 在 n_embd 语义空间里的具体坐标——这是整个模型"理解语言"的基础，绝非无关维度。

进入一个 head 之后，这份 C 维语义并没有消失，只是被投影/压缩了：

Multi-head 的本质: n_embd = head_size × n_head
                            ↑
                    每个 head 只分到一部分语义子空间，
                    但这个子空间仍然来自完整的 C 维语义（W_q/W_k/W_v 各自把全部 C 维都看了一遍再投影）

也就是说，q、k、v 虽然形状都是 (B,T,head_size)，但它们各自都是"C 维语义"经过不同投影矩阵压缩后的产物，不是舍弃了 C 维信息，而是每个 head 用自己的方式重新提炼了一份。

关键的不对称性——q/k 和 v 在语义传递上的角色完全不同：

q、k v
语义去向 参与点积，被压缩成一个标量（wei[t,t']，即相似度分数） 被 wei 加权求和后，直接进入 out，以向量形式保留下来
是否携带到 out 不会——q、k 具体的数值在算出 wei 之后就"用完"了，不会再出现在最终输出里 会——out[t] 本质上就是若干个 v 向量的加权平均，v 的语义内容原封不动地"活"在了输出里

一句话总结：q、k 的使命是"计算该关注谁"，v 的使命是"提供被关注时应该传递什么内容"。 虽然 v 的 shape 看起来只是 head_size（不是完整的 C），但它承载的正是从 n_embd 语义空间投影过来的核心内容——out 最终能表达丰富语义，靠的就是 v，而不是 q、k。
